cmake_minimum_required(VERSION 3.16)
project(livesync_windows_debugger LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This standalone diagnostic requires Windows")
endif()
add_executable(livesync_windows_debugger main.cpp)
target_compile_features(livesync_windows_debugger PRIVATE cxx_std_17)
target_compile_definitions(livesync_windows_debugger PRIVATE NOMINMAX UNICODE _UNICODE)
if(MSVC)
  target_compile_options(livesync_windows_debugger PRIVATE /W4 /WX /EHsc)
endif()
