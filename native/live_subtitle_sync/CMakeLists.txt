cmake_minimum_required(VERSION 3.19)
project(livesync_pcm LANGUAGES CXX)
add_library(livesync_pcm STATIC pcm_buffer.cpp)
target_compile_features(livesync_pcm PUBLIC cxx_std_17)
target_include_directories(livesync_pcm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(livesync_pcm PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(livesync_pcm_bridge SHARED pcm_bridge.cpp)
target_compile_definitions(livesync_pcm_bridge PRIVATE LIVESYNC_PCM_EXPORTS)
target_link_libraries(livesync_pcm_bridge PRIVATE livesync_pcm)
if(MSVC)
  target_compile_options(livesync_pcm PRIVATE /W4 /WX)
else()
  target_compile_options(livesync_pcm PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(livesync_pcm_test pcm_buffer_test.cpp)
  target_link_libraries(livesync_pcm_test PRIVATE livesync_pcm)
  add_test(NAME livesync_pcm_test COMMAND livesync_pcm_test)
endif()
