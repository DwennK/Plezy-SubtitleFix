cmake_minimum_required(VERSION 3.19)
project(livesync_pcm LANGUAGES CXX)
add_library(livesync_pcm STATIC pcm_buffer.cpp)
target_compile_features(livesync_pcm PUBLIC cxx_std_17)
target_include_directories(livesync_pcm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(livesync_pcm PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(livesync_pcm_bridge SHARED pcm_bridge.cpp)
target_compile_definitions(livesync_pcm_bridge PRIVATE LIVESYNC_PCM_EXPORTS)
target_link_libraries(livesync_pcm_bridge PRIVATE livesync_pcm)
set(LIVESYNC_MPV_INCLUDE "" CACHE PATH "Headers from the app's pinned mpv package; enables native capture")
if(LIVESYNC_MPV_INCLUDE)
  find_package(Threads REQUIRED)
  add_library(livesync_capture_bridge SHARED capture_bridge.cpp cpu_capabilities.cpp)
  target_compile_definitions(livesync_capture_bridge PRIVATE LIVESYNC_PCM_EXPORTS)
  target_include_directories(livesync_capture_bridge PRIVATE "${LIVESYNC_MPV_INCLUDE}")
  target_link_libraries(livesync_capture_bridge PRIVATE livesync_pcm Threads::Threads)
  if(MSVC)
    target_compile_options(livesync_capture_bridge PRIVATE /W4 /WX)
  else()
    target_compile_options(livesync_capture_bridge PRIVATE -Wall -Wextra -Werror)
  endif()
endif()
if(MSVC)
  target_compile_options(livesync_pcm PRIVATE /W4 /WX)
else()
  target_compile_options(livesync_pcm PRIVATE -Wall -Wextra -Werror)
endif()

set(LIVESYNC_WHISPER_SOURCE "" CACHE PATH "Exact pinned whisper.cpp checkout; empty disables inference worker")
set(LIVESYNC_CPU_PROFILE "portable" CACHE STRING "portable, guarded-avx2, or avx2-evaluation")
if(LIVESYNC_WHISPER_SOURCE)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/live-subtitle-sync-versions.json" LIVESYNC_MANIFEST)
  string(JSON LIVESYNC_WHISPER_REVISION GET "${LIVESYNC_MANIFEST}" whisper revision)
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIVESYNC_WHISPER_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE LIVESYNC_WHISPER_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE LIVESYNC_GIT_RESULT)
  if(NOT LIVESYNC_GIT_RESULT EQUAL 0 OR NOT LIVESYNC_WHISPER_ACTUAL STREQUAL LIVESYNC_WHISPER_REVISION)
    message(FATAL_ERROR "Whisper source must match the exact LiveSync manifest revision")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static inference dependencies" FORCE)
  # Keep the CPU fallback runnable beyond the machine that built it. Faster
  # runtime-selected variants remain a separate measured packaging decision.
  set(GGML_NATIVE OFF CACHE BOOL "Portable CPU fallback" FORCE)
  foreach(LIVESYNC_X86_OPTION SSE42 AVX AVX2 BMI2 FMA F16C AVX_VNNI AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16)
    set(GGML_${LIVESYNC_X86_OPTION} OFF CACHE BOOL "Portable CPU fallback" FORCE)
  endforeach()
  if(LIVESYNC_CPU_PROFILE STREQUAL "avx2-evaluation" OR LIVESYNC_CPU_PROFILE STREQUAL "guarded-avx2")
    if(NOT WIN32 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|x64)$")
      message(FATAL_ERROR "The AVX2 profile requires Windows x64 and guarded runtime selection")
    endif()
    foreach(LIVESYNC_X86_OPTION SSE42 AVX AVX2 BMI2 FMA F16C)
      set(GGML_${LIVESYNC_X86_OPTION} ON CACHE BOOL "AVX2 variant; runtime capability check required" FORCE)
    endforeach()
  elseif(NOT LIVESYNC_CPU_PROFILE STREQUAL "portable")
    message(FATAL_ERROR "Unknown LiveSync CPU profile")
  endif()
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LIVESYNC_WHISPER_SOURCE}" whisper-build)
  find_package(Threads REQUIRED)
  add_library(livesync_inference STATIC inference_worker.cpp)
  target_compile_features(livesync_inference PUBLIC cxx_std_17)
  target_include_directories(livesync_inference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(livesync_inference PUBLIC livesync_pcm whisper Threads::Threads)
  set_target_properties(livesync_inference PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(livesync_inference_bridge SHARED inference_bridge.cpp)
  target_compile_definitions(livesync_inference_bridge PRIVATE LIVESYNC_INFERENCE_EXPORTS)
  target_link_libraries(livesync_inference_bridge PRIVATE livesync_inference)
  if(MSVC)
    target_compile_options(livesync_inference PRIVATE /W4 /WX)
  else()
    target_compile_options(livesync_inference PRIVATE -Wall -Wextra -Werror)
  endif()
  add_executable(livesync_inference_test inference_worker_test.cpp)
  target_link_libraries(livesync_inference_test PRIVATE livesync_inference)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(livesync_cpu_capabilities_test cpu_capabilities_test.cpp)
  target_compile_features(livesync_cpu_capabilities_test PRIVATE cxx_std_17)
  add_test(NAME livesync_cpu_capabilities_test COMMAND livesync_cpu_capabilities_test)
  add_executable(livesync_pcm_test pcm_buffer_test.cpp)
  target_link_libraries(livesync_pcm_test PRIVATE livesync_pcm)
  add_test(NAME livesync_pcm_test COMMAND livesync_pcm_test)
endif()
